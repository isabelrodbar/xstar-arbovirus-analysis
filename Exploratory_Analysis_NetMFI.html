# Helper function to create beeswarm plot for a single analyte (by plate, colored by Wave)
plot_sample_mfi_beeswarm <- function(data, analyte_name) {
  # Filter data for this analyte and remove non-positive MFI values for log scale
  plot_data <- data %>%
    filter(Analyte == analyte_name, !is.na(MFI_Net), MFI_Net > 0) %>%
    mutate(
      # Create ordered factor for WavePlate based on Wave and plate number
      WavePlate = factor(WavePlate, levels = unique(WavePlate[order(Wave, as.numeric(PlateNumber_Clean))])),
      SampleType = factor(SampleType, levels = c("Normal", "PBT", "Faint Eluent", "Faint"))
    )

  # Calculate summary statistics
  n_samples <- n_distinct(plot_data$Barcode)
  median_mfi <- median(plot_data$MFI_Net, na.rm = TRUE)
  n_flagged <- sum(plot_data$LowCount_Flag, na.rm = TRUE)
  pct_flagged <- 100 * mean(plot_data$LowCount_Flag, na.rm = TRUE)

  # Count special sample types
  n_special <- sum(plot_data$SampleType != "Normal")

  # Count by wave
  wave_counts <- plot_data %>% distinct(Barcode, Wave) %>% count(Wave)
  wave_text <- paste(sprintf("%s: %d", wave_counts$Wave, wave_counts$n), collapse = " | ")

  # Create plot with Wave coloring
  p <- ggplot(plot_data, aes(x = WavePlate, y = MFI_Net)) +
    geom_beeswarm(aes(color = Wave, shape = LowCount_Flag), alpha = 0.7, cex = 1.5, size = 1.5) +
    geom_boxplot(alpha = 0.2, outlier.shape = NA, width = 0.5) +
    scale_color_manual(
      values = c('Wave 3' = '#E41A1C', 'Wave 5' = '#377EB8'),
      name = 'Wave'
    ) +
    scale_shape_manual(
      values = c('FALSE' = 16, 'TRUE' = 4),
      labels = c('FALSE' = 'Count OK', 'TRUE' = 'Count Low'),
      name = 'Bead Count'
    ) +
    scale_y_log10() +
    labs(
      title = analyte_name,
      subtitle = sprintf("N = %d samples (%s) | Net MFI = %.0f | %.1f%% low count",
                         n_samples, wave_text, median_mfi, pct_flagged),
      x = 'Wave-Plate',
      y = 'MFI (Net, log scale)'
    ) +
    theme_bw(base_size = 11) +
    theme(
      legend.position = 'bottom',
      plot.title = element_text(size = 12, face = "bold"),
      plot.subtitle = element_text(size = 9, color = "gray30"),
      axis.text.x = element_text(angle = 45, hjust = 1)
    )

  return(p)
}

# Helper function to create single panel plot for a disease group (with Wave coloring)
plot_disease_group_panel <- function(data, analyte_list, group_title) {
  # Filter data for these analytes and remove non-positive MFI values for log scale
  plot_data <- data %>%
    filter(Analyte %in% analyte_list, !is.na(MFI_Net), MFI_Net > 0) %>%
    mutate(
      # Clean analyte names for display
      Analyte_Label = str_replace(Analyte, "ARB_|MAL_|NTD_|CHO_|HEP_|VPD_|HHV_|CTRL_", ""),
      Analyte_Label = str_replace(Analyte_Label, "SNAP_", ""),
      Analyte_Label = factor(Analyte_Label, levels = unique(Analyte_Label)),
      SampleType = factor(SampleType, levels = c("Normal", "PBT", "Faint Eluent", "Faint"))
    )

  # Calculate summary statistics
  n_samples <- n_distinct(plot_data$Barcode)
  n_analytes <- length(analyte_list)
  median_mfi <- median(plot_data$MFI_Net, na.rm = TRUE)
  n_special <- sum(plot_data$SampleType != "Normal")

  # Count by wave
  wave_counts <- plot_data %>% distinct(Barcode, Wave) %>% count(Wave)
  wave_text <- paste(sprintf("%s: %d", wave_counts$Wave, wave_counts$n), collapse = " | ")

  # Create single panel plot with analyte on x-axis, colored by Wave
  p <- ggplot(plot_data, aes(x = Analyte_Label, y = MFI_Net)) +
    geom_beeswarm(aes(color = Wave, shape = LowCount_Flag), alpha = 0.7, cex = 0.8, size = 1.2) +
    geom_boxplot(alpha = 0.2, outlier.shape = NA, width = 0.5) +
    scale_color_manual(
      values = c('Wave 3' = '#E41A1C', 'Wave 5' = '#377EB8'),

    scale_shape_manual(

      labels = c('FALSE' = 'Count OK', 'TRUE' = 'Count Low'),
      name = 'Bead Count'
    ) +
    scale_y_log10() +
    labs(
      title = group_title,
      subtitle = sprintf("N = %d samples (%s) | %d antigens | Net MFI = %.0f",
                         n_samples, wave_text, n_analytes, median_mfi),
      x = 'Antigen',
      y = 'MFI (Net, log scale)'
    ) +
    theme_bw(base_size = 11) +
    theme(
      legend.position = 'bottom',
      plot.title = element_text(size = 12, face = "bold"),
      plot.subtitle = element_text(size = 9, color = "gray30"),
      axis.text.x = element_text(angle = 45, hjust = 1, size = 9)
    )

  return(p)
}

# Helper function to plot SNAP ratio by plate
plot_snap_ratio_beeswarm <- function(data, analyte_name) {
  # Filter data for this analyte and remove non-finite values
  plot_data <- data %>%
    filter(Analyte == analyte_name, !is.na(MFI_Net_SNAP_Ratio), is.finite(MFI_Net_SNAP_Ratio), MFI_Net_SNAP_Ratio > 0) %>%
    mutate(
      # Create ordered factor for WavePlate
      WavePlate = factor(WavePlate, levels = unique(WavePlate[order(Wave, as.numeric(PlateNumber_Clean))])),
      SampleType = factor(SampleType, levels = c("Normal", "PBT", "Faint Eluent", "Faint"))
    )

  # Calculate summary statistics
  n_samples <- n_distinct(plot_data$Barcode)
  median_ratio <- median(plot_data$MFI_Net_SNAP_Ratio, na.rm = TRUE)

  # Count by wave
  wave_counts <- plot_data %>% distinct(Barcode, Wave) %>% count(Wave)
  wave_text <- paste(sprintf("%s: %d", wave_counts$Wave, wave_counts$n), collapse = " | ")

  # Create plot
  p <- ggplot(plot_data, aes(x = WavePlate, y = MFI_Net_SNAP_Ratio)) +
    geom_hline(yintercept = 1, linetype = "dashed", color = "gray50", size = 0.5) +

      title = paste0(analyte_name, " - SNAP Ratio"),
